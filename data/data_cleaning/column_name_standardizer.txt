import sqlite3
import pandas as pd
import ast

DB_PATH = "data/db/military_equipment_01-Copy.db"
MASTER_LIST = "a_meta_table"


def fix_tuple_column(col):
    """
    Handles tuple-like column names:
    - real tuples
    - stringified tuples
    - quoted tuples
    Returns (base, section) or (col, None)
    """

    # Case 1: real tuple = return both the coloumn name and section name
    if isinstance(col, tuple) and len(col) == 2:
        return str(col[0]), str(col[1])

    # Case 2: string that looks like a tuple
    if isinstance(col, str):
        try:
            parsed = ast.literal_eval(col)
            if isinstance(parsed, tuple) and len(parsed) == 2:
                return str(parsed[0]), str(parsed[1])
        except:
            print(f"passing on {col}")
            pass

    # Not a tuple
    return str(col), None


def normalize_column_string(col):
    """
    Ensures column names are clean strings:
    - removes parentheses
    - removes stray quotes
    - strips whitespace
    """

    col = str(col)
    col = col.strip()
    col = col.replace("(", "").replace(")", "")
    col = col.replace("'", "").replace('"', "")
    return col


def clean_column_names(df):
    """
    Cleans column names and inserts section rows when needed.
    """

    new_cols = []
    section_values = {}   # col_index -> section label

    # Pass 1: clean column names and detect sections
    for idx, col in enumerate(df.columns):
        base, section = fix_tuple_column(col)
        base = normalize_column_string(base)

        new_cols.append(base)

        if section:
            section_values[idx] = section

    df.columns = new_cols

    # Pass 2: insert section row if needed
    if section_values:
        section_row = [None] * len(new_cols)
        for idx, sec in section_values.items():
            section_row[idx] = sec
        
        print(f"added row of {section_row[0]}")

        df.loc[-1] = section_row
        df.index = df.index + 1
        df.sort_index(inplace=True)

    return df


def fix_column_names():
    conn = sqlite3.connect(DB_PATH)
    cursor = conn.cursor()

    cursor.execute(f"SELECT table_name FROM {MASTER_LIST}")
    tables = [row[0] for row in cursor.fetchall()]

    for table in tables:
        print(f"Cleaning headers for: {table}")

        try:
            df = pd.read_sql_query(f"SELECT * FROM '{table}'", conn)
            df = clean_column_names(df)
            df.to_sql(table, conn, if_exists="replace", index=False)

        except Exception as e:
            print(f"Error cleaning {table}: {e}")

    conn.close()
    print("Header cleaning complete.")


if __name__ == "__main__":
    fix_column_names()
